// Define the function first
function convertDaisyUITheme(themeString, themeName) {
  // Extract just the CSS variables and properties from the DaisyUI theme
  const lines = themeString.split('\n');
  let cssVars = [];
  
  // Mapping from DaisyUI long names to short Tailwind names
  const colorMapping = {
    '--color-base-100': '--b1',
    '--color-base-200': '--b2', 
    '--color-base-300': '--b3',
    '--color-base-content': '--bc',
    '--color-primary': '--p',
    '--color-primary-content': '--pc',
    '--color-secondary': '--s',
    '--color-secondary-content': '--sc',
    '--color-accent': '--a',
    '--color-accent-content': '--ac',
    '--color-neutral': '--n',
    '--color-neutral-content': '--nc',
    '--color-info': '--in',
    '--color-info-content': '--inc',
    '--color-success': '--su',
    '--color-success-content': '--suc',
    '--color-warning': '--wa',
    '--color-warning-content': '--wac',
    '--color-error': '--er',
    '--color-error-content': '--erc'
  };
  
  for (let line of lines) {
    line = line.trim();
    
    // Skip non-CSS variable lines
    if (!line.startsWith('--')) continue;
    
    // Extract property and value
    const colonIndex = line.indexOf(':');
    if (colonIndex === -1) continue;
    
    let property = line.substring(0, colonIndex).trim();
    let value = line.substring(colonIndex + 1).trim();
    
    // Remove semicolon if present
    if (value.endsWith(';')) {
      value = value.slice(0, -1);
    }
    
    // Convert long color names to short ones
    if (colorMapping[property]) {
      property = colorMapping[property];
    }
    
    // Remove oklch() wrapper and keep just the values
    if (value.startsWith('oklch(') && value.endsWith(')')) {
      value = value.slice(6, -1); // Remove 'oklch(' and ')'
    }
    
    cssVars.push(`  ${property}: ${value};`);
  }
  
  // Generate the complete CSS theme
  const css = `/* Scoped DaisyUI theme: ${themeName} */
#eveve-widget[data-theme="${themeName}"] {
${cssVars.join('\n')}

  --rounded-box: 0.5rem;
  --rounded-btn: 0.25rem;
  --border-btn: 1px;
}`;
  
  return css;
}
